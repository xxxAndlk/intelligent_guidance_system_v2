## 一、项目概述

### 1.1 迁移目标

将现有Java SpringBoot单体应用迁移到Go+Kratos微服务架构，并引入RAG检索+Agent智能体能力。

### 1.2 基础设施约束

- **数据库**: MySQL 8.0
- **缓存**: Redis 7.x
- **消息队列**: RabbitMQ
- **服务注册**: Consul
- **向量数据库**: Qdrant（新增）
- **LLM服务**: DeepSeek API

### 5.1 AI服务核心实现

```go
// service/ai/internal/service/ai.go
package service

import (
    "context"
    
    "github.com/go-kratos/kratos/v2/log"
    pb "medical-system/api/ai/v1"
    "medical-system/ai/internal/biz"
)

// AIService AI服务实现
type AIService struct {
    pb.UnimplementedAIServiceServer
    
    uc  *biz.DiagnosisUsecase
    log *log.Helper
}

// NewAIService 创建AI服务
func NewAIService(uc *biz.DiagnosisUsecase, logger log.Logger) *AIService {
    return &AIService{
        uc:  uc,
        log: log.NewHelper(logger),
    }
}

// CreateDiagnosisSession 创建导诊会话
func (s *AIService) CreateDiagnosisSession(ctx context.Context, req *pb.CreateDiagnosisSessionRequest) (*pb.DiagnosisSession, error) {
    s.log.Infof("创建导诊会话: patient_id=%d", req.PatientId)
    
    session, err := s.uc.CreateSession(ctx, req.PatientId)
    if err != nil {
        s.log.Errorf("创建会话失败: %v", err)
        return nil, err
    }
    
    return convertSessionToProto(session), nil
}

// AddSymptom 添加症状
func (s *AIService) AddSymptom(ctx context.Context, req *pb.AddSymptomRequest) (*pb.DiagnosisSession, error) {
    s.log.Infof("添加症状: session_id=%s, body_part=%s", req.SessionId, req.BodyPart)
    
    session, err := s.uc.AddSymptom(ctx, req.SessionId, req.BodyPart, req.Description, int32(req.Severity))
    if err != nil {
        s.log.Errorf("添加症状失败: %v", err)
        return nil, err
    }
    
    return convertSessionToProto(session), nil
}

// GenerateRecommendation 生成推荐（核心接口）
func (s *AIService) GenerateRecommendation(ctx context.Context, req *pb.GenerateRecommendationRequest) (*pb.Recommendation, error) {
    s.log.Infof("生成推荐: session_id=%s, use_vector=%v, use_agent=%v", 
        req.SessionId, req.UseVectorSearch, req.UseAgent)
    
    rec, err := s.uc.GenerateRecommendation(ctx, biz.GenerateRecommendationRequest{
        SessionID:      req.SessionId,
        UseVectorSearch: req.UseVectorSearch,
        VectorLimit:    int(req.VectorLimit),
        UseAgent:       req.UseAgent,
    })
    if err != nil {
        s.log.Errorf("生成推荐失败: %v", err)
        return nil, err
    }
    
    return convertRecommendationToProto(rec), nil
}

// ChatStream 流式对话
func (s *AIService) ChatStream(stream pb.AIService_ChatStreamServer) error {
    ctx := stream.Context()
    
    for {
        msg, err := stream.Recv()
        if err != nil {
            return err
        }
        
        // 处理消息并流式返回
        response, err := s.uc.ProcessChatMessage(ctx, msg.SessionId, msg.Content)
        if err != nil {
            return err
        }
        
        if err := stream.Send(&pb.ChatMessage{
            SessionId: msg.SessionId,
            Role:      pb.ConversationRole_ROLE_ASSISTANT,
            Content:   response,
        }); err != nil {
            return err
        }
    }
}

// 转换函数
func convertSessionToProto(session *aggregate.DiagnosisSession) *pb.DiagnosisSession {
    pbSession := &pb.DiagnosisSession{
        Id:        string(session.ID),
```

```go
// GenerateRecommendation 生成推荐（核心逻辑）
func (uc *DiagnosisUsecase) GenerateRecommendation(ctx context.Context, req GenerateRecommendationRequest) (*vo.Recommendation, error) {
    // 1. 获取会话
    session, err := uc.sessionRepo.GetByID(ctx, req.SessionID)
    if err != nil {
        return nil, err
    }
    
    // 2. 检查是否可以生成推荐
    if !session.IsReadyForAnalysis() {
        return nil, errors.New("症状信息不足，无法生成推荐")
    }
    
    // 3. 准备RAG上下文
    context := session.GetContextForRAG()
    
    var recommendation vo.Recommendation
    
    if req.UseAgent {
        // 4. 使用Agent推理（完整ReAct模式）
        recommendation, err = uc.generateWithAgent(ctx, session, context, req)
    } else {
        // 5. 使用规则+向量检索（混合模式）
        recommendation, err = uc.generateWithHybrid(ctx, session, context, req)
    }
    
    if err != nil {
        return nil, err
    }
    
    // 6. 更新会话
    if err := session.GenerateRecommendation(recommendation); err != nil {
        return nil, err
    }
    
    // 7. 保存
    if err := uc.sessionRepo.Update(ctx, session); err != nil {
        return nil, err
    }
    
    return &recommendation, nil
}

// generateWithHybrid 使用混合模式生成推荐
func (uc *DiagnosisUsecase) generateWithHybrid(ctx context.Context, session *aggregate.DiagnosisSession, context string, req GenerateRecommendationRequest) (vo.Recommendation, error) {
    var rec vo.Recommendation
    
    // 1. 向量化查询
    queryVector, err := uc.embedder.Embed(ctx, context)
    if err != nil {
        return rec, fmt.Errorf("向量化失败: %w", err)
    }
    
    // 2. 向量检索疾病
    if req.UseVectorSearch {
        diseaseResults, err := uc.vectorDBACL.SearchDiseases(ctx, queryVector, req.VectorLimit)
        if err != nil {
            uc.log.Warnf("疾病向量检索失败: %v", err)
        } else {
            session.RecordVectorQuery(context, entity.VectorSourceDisease, diseaseResults)
            
            // 转换为推荐
            for _, result := range diseaseResults {
                rec.SuspectedDiseases = append(rec.SuspectedDiseases, vo.DiseaseRecommendation{
                    DiseaseID:   result.ID,
                    DiseaseName: result.Payload["name"],
                    Symptoms:    result.Payload["symptoms"],
                    Diagnosis:   result.Payload["diagnosis"],
                    MatchScore:  result.Score,
                    Probability: result.Score, // 使用相似度作为概率
                })
            }
        }
    }
    
    // 3. 向量检索医生
    doctorResults, err := uc.vectorDBACL.SearchDoctors(ctx, queryVector, 4)
    if err != nil {
        uc.log.Warnf("医生向量检索失败: %v", err)
    } else {
        for _, result := range doctorResults {
            rec.RecommendedDoctors = append(rec.RecommendedDoctors, vo.DoctorRecommendation{
                DoctorID:     result.ID,
                DoctorName:   result.Payload["name"],
                Title:        result.Payload["title"],
                IsExpert:     result.Payload["is_expert"] == "true",
                DepartmentID: parseInt64(result.Payload["department_id"]),
                Intro:        result.Payload["intro"],
                MatchScore:   result.Score,
            })
        }
    }
    
    // 4. 向量检索药品
    drugResults, err := uc.vectorDBACL.SearchDrugs(ctx, queryVector, 4)
    if err != nil {
        uc.log.Warnf("药品向量检索失败: %v", err)
    } else {
        for _, result := range drugResults {
            rec.RecommendedDrugs = append(rec.RecommendedDrugs, vo.DrugRecommendation{
                DrugID:      result.ID,
                DrugName:    result.Payload["name"],
                Description: result.Payload["description"],
                Category:    result.Payload["category"],
                Usage:       result.Payload["usage"],
            })
        }
    }
    
    // 5. 使用LLM生成分析说明
    analysis, err := uc.generateAnalysis(ctx, context, rec)
    if err != nil {
        uc.log.Warnf("生成分析说明失败: %v", err)
        rec.Analysis = "基于您的症状描述，我们为您推荐了可能的疾病、适合就诊的科室和医生。建议尽快就医进行专业诊断。"
    } else {
        rec.Analysis = analysis
    }
    
    rec.Confidence = uc.calculateConfidence(rec)
    rec.GeneratedAt = time.Now().Unix()
    
    return rec, nil
}

// generateWithAgent 使用Agent生成推荐
func (uc *DiagnosisUsecase) generateWithAgent(ctx context.Context, session *aggregate.DiagnosisSession, context string, req GenerateRecommendationRequest) (vo.Recommendation, error) {
    // 使用ReAct Agent进行多步推理
    result, err := uc.agent.Execute(ctx, agent.Task{
        Type:    "diagnosis",
        Context: context,
        Skills:  []string{"collect_symptom", "search_disease", "search_doctor", "search_drug", "analyze_symptom"},
    })
    
    if err != nil {
        return vo.Recommendation{}, err
    }
    
    return result.Recommendation, nil
}

// generateAnalysis 生成分析说明
func (uc *DiagnosisUsecase) generateAnalysis(ctx context.Context, context string, rec vo.Recommendation) (string, error) {
    prompt := fmt.Sprintf(`
基于以下症状和推荐结果，生成一段专业的分析说明：

症状描述：
%s

推荐疾病：%v
推荐医生：%v

请用中文生成一段200字以内的分析说明，解释为什么推荐这些疾病和医生。
`, context, rec.SuspectedDiseases, rec.RecommendedDoctors)

    return uc.llmClient.Complete(ctx, prompt)
}

// calculateConfidence 计算置信度
func (uc *DiagnosisUsecase) calculateConfidence(rec vo.Recommendation) float64 {
    if len(rec.SuspectedDiseases) == 0 {
        return 0.5
    }
    
    // 使用最高匹配分数作为置信度
    maxScore := 0.0
    for _, d := range rec.SuspectedDiseases {
        if d.MatchScore > maxScore {
            maxScore = d.MatchScore
        }
    }
    
    return maxScore
}
```

```go
// SiliconFlowEmbedder SiliconFlow Embedding实现
type SiliconFlowEmbedder struct {

    model  string
    client *http.Client

// NewSiliconFlowEmbedder 创建Embedder
func NewSiliconFlowEmbedder(apiKey string) *SiliconFlowEmbedder {

        apiURL: "https://api.siliconflow.cn/v1/embeddings",
        model:  "BAAI/bge-large-zh-v1.5",

    }
}

// Embed 单文本向量化
func (e *SiliconFlowEmbedder) Embed(ctx context.Context, text string) ([]float32, error) {
    results, err := e.EmbedBatch(ctx, []string{text})
    if err != nil {
        return nil, err
    }
    if len(results) == 0 {
        return nil, errors.New("empty embedding result")
    }
    return results[0], nil
}

// EmbedBatch 批量向量化
func (e *SiliconFlowEmbedder) EmbedBatch(ctx context.Context, texts []string) ([][]float32, error) {
    reqBody := map[string]interface{}{
        "model": e.model,
        "input": texts,
    }
    
    jsonData, err := json.Marshal(reqBody)
    if err != nil {
        return nil, err
    }
    
    req, err := http.NewRequestWithContext(ctx, "POST", e.apiURL, strings.NewReader(string(jsonData)))
    if err != nil {
        return nil, err
    }
    
    req.Header.Set("Content-Type", "application/json")
    req.Header.Set("Authorization", "Bearer "+e.apiKey)
    
    resp, err := e.client.Do(req)
    if err != nil {
        return nil, err
    }
    defer resp.Body.Close()
    
    if resp.StatusCode != http.StatusOK {
        return nil, fmt.Errorf("embedding API returned status %d", resp.StatusCode)
    }
    
    var result struct {
        Data []struct {
            Embedding []float32 `json:"embedding"`
        } `json:"data"`
    }
    
    if err := json.NewDecoder(resp.Body).Decode(&result); err != nil {
        return nil, err
    }
    
    embeddings := make([][]float32, len(result.Data))
    for i, d := range result.Data {
        embeddings[i] = d.Embedding
    }
    
    return embeddings, nil
}
```

```go
// search 通用搜索方法
func (r *QdrantRepository) search(ctx context.Context, collection string, queryVector []float32, filter *qdrant.Filter, limit int) ([]entity.VectorResult, error) {
    req := &qdrant.QueryPoints{
        CollectionName: collection,
        Query: &qdrant.Query{
            Variant: &qdrant.Query_Nearest{
                Nearest: &qdrant.VectorInput{
                    Variant: &qdrant.VectorInput_Vector{
                        Vector: &qdrant.Vector{
                            Data: queryVector,
                        },
                    },
                },
            },
        },
        Filter:      filter,
        Limit:       uint64(limit),
        WithPayload: &qdrant.WithPayloadSelector{
            SelectorOptions: &qdrant.WithPayloadSelector_Enable{
                Enable: true,
            },
        },
        WithVectors: &qdrant.WithVectorsSelector{
            SelectorOptions: &qdrant.WithVectorsSelector_Enable{
                Enable: false,
            },
        },
    }
    
    resp, err := r.client.Query(ctx, req)
    if err != nil {
        r.log.Errorf("Qdrant搜索失败 [collection=%s]: %v", collection, err)
        return nil, fmt.Errorf("vector search failed: %w", err)
    }
    
    var results []entity.VectorResult
    for _, point := range resp {
        payload := make(map[string]string)
        for k, v := range point.Payload {
            if sv, ok := v.Kind.(*qdrant.Value_StringValue); ok {
                payload[k] = sv.StringValue
            }
        }
        
        results = append(results, entity.VectorResult{
            ID:      point.Id.GetNum(),
            Score:   point.Score,
            Payload: payload,
            Source:  entity.VectorSource(collection),
        })
    }
    
    r.log.Infof("Qdrant搜索成功 [collection=%s, results=%d]", collection, len(results))
    
    return results, nil
}

// Upsert 插入或更新向量
func (r *QdrantRepository) Upsert(ctx context.Context, collection string, doc entity.VectorDocument) error {
    // 转换metadata
    payload := make(map[string]*qdrant.Value)
    for k, v := range doc.Metadata {
        payload[k] = &qdrant.Value{
            Kind: &qdrant.Value_StringValue{
                StringValue: v,
            },
        }
    }
    
    req := &qdrant.UpsertPoints{
        CollectionName: collection,
        Points: []*qdrant.PointStruct{
            {
                Id: &qdrant.PointId{
                    PointIdOptions: &qdrant.PointId_Num{
                        Num: doc.ID,
                    },
                },
                Vectors: &qdrant.Vectors{
                    VectorsOptions: &qdrant.Vectors_Vector{
                        Vector: &qdrant.Vector{
                            Data: doc.Vector,
                        },
                    },
                },
                Payload: payload,
            },
        },
    }
    
    _, err := r.client.Upsert(ctx, req)
    if err != nil {
        r.log.Errorf("Qdrant插入失败 [collection=%s, id=%d]: %v", collection, doc.ID, err)
        return fmt.Errorf("upsert failed: %w", err)
    }
    
    return nil
}
```

### 7.2 ReAct Agent实现

```go
// service/ai/internal/agent/react.go
package agent

import (
    "context"
    "encoding/json"
    "fmt"
    "strings"
    
    "github.com/go-kratos/kratos/v2/log"
    "medical-system/ai/internal/service/skills"

// ReActAgent ReAct推理Agent
type ReActAgent struct {
    BaseAgent
    log *log.Helper
}

// NewReActAgent 创建ReAct Agent
func NewReActAgent(llmClient LLMClient, skills map[string]skills.Skill, logger log.Logger) *ReActAgent {
    return &ReActAgent{
        BaseAgent: BaseAgent{
            llmClient: llmClient,
            skills:    skills,
            maxSteps:  5,
        },
        log: log.NewHelper(logger),
    }
}

// Execute 执行任务
func (a *ReActAgent) Execute(ctx context.Context, task Task) (Result, error) {
    var result Result
    var steps []Step
    
    context := task.Context
    availableSkills := a.getAvailableSkills(task.Skills)
    
    for stepNum := 1; stepNum <= a.maxSteps; stepNum++ {
        // 1. 思考
        thought, err := a.think(ctx, context, availableSkills, steps)
        if err != nil {
            return result, err
        }
        
        // 2. 行动
        action, input, isFinal, err := a.act(ctx, thought, availableSkills)
        if err != nil {
            return result, err
        }
        
        step := Step{
            Number:  stepNum,
            Thought: thought,
            Action:  action,
            Input:   input,
            IsFinal: isFinal,
        }
        
        // 如果是最终答案
        if isFinal {
            step.Output = action
            steps = append(steps, step)
            result.Answer = action
            result.Steps = steps
            break
        }
        
        // 3. 执行工具
        output, err := a.executeSkill(ctx, action, input)
        if err != nil {
            step.Output = fmt.Sprintf("Error: %v", err)
        } else {
            step.Output = output
        }
        
        steps = append(steps, step)
        
        // 更新上下文
        context = fmt.Sprintf("%s\n\nStep %d:\nThought: %s\nAction: %s[%s]\nObservation: %s",
            context, stepNum, thought, action, input, step.Output)
    }
    
    // 生成推荐结果
    recommendation, err := a.generateRecommendation(ctx, context, steps)
    if err != nil {
        a.log.Warnf("生成推荐失败: %v", err)
    }
    result.Recommendation = recommendation
    
    return result, nil
}

// think 思考步骤
func (a *ReActAgent) think(ctx context.Context, context string, skills []skills.Skill, history []Step) (string, error) {
    prompt := a.buildThinkPrompt(context, skills, history)
    
    response, err := a.llmClient.Complete(ctx, prompt)
    if err != nil {
        return "", err
    }
    
    return response, nil
}

// act 行动步骤
func (a *ReActAgent) act(ctx context.Context, thought string, availableSkills []skills.Skill) (action, input string, isFinal bool, err error) {
    prompt := fmt.Sprintf(`
基于以下思考，决定下一步行动：

思考：%s

可用工具：
%s

请用以下格式响应：
如果继续使用工具：
Action: 工具名称
Input: 工具输入（JSON格式）

如果已有最终答案：
Final Answer: 最终答案
`, thought, a.formatSkills(availableSkills))

    response, err := a.llmClient.Complete(ctx, prompt)
    if err != nil {
        return "", "", false, err
    }
    
    // 解析响应
    if strings.Contains(response, "Final Answer:") {
        answer := strings.TrimPrefix(response, "Final Answer:")
        answer = strings.TrimSpace(answer)
        return answer, "", true, nil
    }
    
    // 解析Action和Input
    lines := strings.Split(response, "\n")
    for _, line := range lines {
        if strings.HasPrefix(line, "Action:") {
            action = strings.TrimSpace(strings.TrimPrefix(line, "Action:"))
        }
        if strings.HasPrefix(line, "Input:") {
            input = strings.TrimSpace(strings.TrimPrefix(line, "Input:"))
        }
    }
    
    return action, input, false, nil
}

// executeSkill 执行技能
func (a *ReActAgent) executeSkill(ctx context.Context, name, input string) (string, error) {
    skill, exists := a.skills[name]
    if !exists {
        return "", fmt.Errorf("skill not found: %s", name)
    }
    
    var params map[string]interface{}
    if err := json.Unmarshal([]byte(input), &params); err != nil {
        return "", err
    }
    
    return skill.Execute(ctx, params)
}

// buildThinkPrompt 构建思考提示词
func (a *ReActAgent) buildThinkPrompt(context string, skills []skills.Skill, history []Step) string {
    var historyStr string
    for _, step := range history {
        historyStr += fmt.Sprintf("\nStep %d: %s -> %s", step.Number, step.Action, step.Output)
    }
    
    return fmt.Sprintf(`
你是一个医疗导诊助手，正在帮助患者分析症状并推荐合适的科室和医生。

当前任务上下文：
%s

历史步骤：%s

可用工具：
%s

请思考下一步应该做什么：
1. 是否需要收集更多症状信息？
2. 是否需要搜索相关疾病？
3. 是否需要搜索合适的医生？
4. 是否已有足够信息给出推荐？

请用简短的一句话描述你的思考。`, context, historyStr, a.formatSkills(skills))
}

// formatSkills 格式化技能列表
func (a *ReActAgent) formatSkills(skills []skills.Skill) string {
    var sb strings.Builder
    for _, s := range skills {
        sb.WriteString(fmt.Sprintf("- %s: %s\n", s.Name(), s.Description()))
    }
    return sb.String()
}

// getAvailableSkills 获取可用技能
func (a *ReActAgent) getAvailableSkills(names []string) []skills.Skill {
    var result []skills.Skill
    for _, name := range names {
        if skill, exists := a.skills[name]; exists {
            result = append(result, skill)
        }
    }
    return result
}
```
