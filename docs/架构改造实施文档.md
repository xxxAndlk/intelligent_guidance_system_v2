## 一、项目概述

### 1.1 改造目标

将现有智慧医疗系统从传统三端分离架构升级为**MVC增强架构**，实现：
- ✅ 统一软删除机制
- ✅ 多角色权限系统
- ✅ 代码复用优化
- ✅ 数据权限控制

### 1.2 改造范围

| 层级 | 改造内容 |
|-----|---------|
| 实体层 | 15个实体类统一继承BaseEntity |
| 权限层 | 三端JWT统一，新增角色权限表 |
| Controller层 | 提取BaseController，减少重复代码 |
| Service层 | 新增PermissionService |
| 数据库 | 15张表添加deleted_at字段，新增权限相关表 |

### 1.3 改造计划

```
Week 1:
Day 1-2: Phase 1 - 实体类软删除改造
```

#### 2.2.1 创建BaseEntity.java

```java
package com.example.pojo.entity;

import com.baomidou.mybatisplus.annotation.*;
import lombok.Data;

import java.time.LocalDateTime;

/**
 * 实体基类
 * 统一包含：主键、创建时间、更新时间、逻辑删除字段
 */
@Data
public abstract class BaseEntity {
    
    /**
     * 主键ID
     */
    @TableId(type = IdType.AUTO)
    private Long id;
    
    /**
     * 创建时间
     */
    @TableField(fill = FieldFill.INSERT)
    private LocalDateTime createTime;
    
    /**
     * 更新时间
     */
    @TableField(fill = FieldFill.INSERT_UPDATE)
    private LocalDateTime updateTime;
    
    /**
     * 逻辑删除时间
     * null = 未删除
     * 时间值 = 已删除（删除时间）
     */
    @TableLogic(value = "null", delval = "now()")
    @TableField(fill = FieldFill.INSERT)
    private LocalDateTime deletedAt;
}
```

### 2.3 MyBatis-Plus配置

#### 2.3.1 创建MybatisPlusConfig.java

```java
package com.example.config;

import com.baomidou.mybatisplus.annotation.DbType;
import com.baomidou.mybatisplus.core.handlers.MetaObjectHandler;
import com.baomidou.mybatisplus.extension.plugins.MybatisPlusInterceptor;
import com.baomidou.mybatisplus.extension.plugins.inner.PaginationInnerInterceptor;
import lombok.extern.slf4j.Slf4j;
import org.apache.ibatis.reflection.MetaObject;
import org.springframework.context.annotation.Bean;
import org.springframework.context.annotation.Configuration;

import java.time.LocalDateTime;

/**
 * MyBatis-Plus配置类
 */
@Configuration
@Slf4j
public class MybatisPlusConfig {

    /**
     * MyBatis-Plus插件配置
     */
    @Bean
    public MybatisPlusInterceptor mybatisPlusInterceptor() {
        MybatisPlusInterceptor interceptor = new MybatisPlusInterceptor();
        // 分页插件
        interceptor.addInnerInterceptor(new PaginationInnerInterceptor(DbType.MYSQL));
        // 逻辑删除插件（自动添加deleted_at IS NULL条件）
        // MyBatis-Plus 3.5+ 自动识别 @TableLogic，无需额外配置
        return interceptor;
    }

    /**
     * 自动填充处理器
     */
    @Bean
    public MetaObjectHandler metaObjectHandler() {
        return new MetaObjectHandler() {
            @Override
            public void insertFill(MetaObject metaObject) {
                log.debug("开始插入填充...");
                // 创建时间
                this.strictInsertFill(metaObject, "createTime", LocalDateTime.class, LocalDateTime.now());
                // 更新时间
                this.strictInsertFill(metaObject, "updateTime", LocalDateTime.class, LocalDateTime.now());
                // 逻辑删除字段默认为null
            }

            @Override
            public void updateFill(MetaObject metaObject) {
                log.debug("开始更新填充...");
                // 更新时间
                this.strictUpdateFill(metaObject, "updateTime", LocalDateTime.class, LocalDateTime.now());
            }
        };
    }
}
```

#### 3.3.1 统一用户上下文

```java
package com.example.common.context;

import com.example.common.enumeration.UserType;
import lombok.Data;

import java.io.Serializable;
import java.util.List;

/**
 * 统一用户上下文
 * 支持：患者、医生（多角色）、管理员
 */
@Data
public class UserContext implements Serializable {
    
    private static final long serialVersionUID = 1L;
    
    /**
     * 用户ID
     */
    private Long userId;
    
    /**
     * 用户类型
     */
    private UserType userType;
    
    /**
     * 用户名
     */
    private String username;
    
    /**
     * 真实姓名
     */
    private String realName;
    
    /**
     * 角色列表
     */
    private List<String> roles;
    
    /**
     * 权限列表
     */
    private List<String> permissions;
    
    /**
     * 所属科室ID（医生）
     */
    private Long deptId;
    
    /**
     * 管理的科室ID列表（科室主任等）
     */
    private List<Long> managedDeptIds;
    
    /**
     * JWT令牌
     */
    private String token;
    
    // ==================== 便捷方法 ====================
    
    /**
     * 是否包含指定角色
     */
    public boolean hasRole(String role) {
        return roles != null && roles.contains(role);
    }
    
    /**
     * 是否包含任意角色
     */
    public boolean hasAnyRole(String... roles) {
        if (this.roles == null) return false;
        for (String role : roles) {
            if (this.roles.contains(role)) return true;
        }
        return false;
    }
    
    /**
     * 是否有权限
     */
    public boolean hasPermission(String permission) {
        return permissions != null && permissions.contains(permission);
    }
    
    /**
     * 是否能访问指定科室
     */
    public boolean canAccessDept(Long deptId) {
        if (this.deptId != null && this.deptId.equals(deptId)) {
            return true;
        }
        if (managedDeptIds != null) {
            return managedDeptIds.contains(deptId);
        }
        return false;
    }
    
    /**
     * 是否是管理员
     */
    public boolean isAdmin() {
        return userType == UserType.ADMIN || hasRole("SUPER_ADMIN");
    }
    
    /**
     * 是否是医生
     */
    public boolean isDoctor() {
        return userType == UserType.DOCTOR;
    }
    
    /**
     * 是否是患者
     */
    public boolean isPatient() {
        return userType == UserType.PATIENT;
    }
}
```

#### 3.3.4 统一JWT拦截器

```java
package com.example.interceptor;

import com.example.common.constant.JwtClaimsConstant;
import com.example.common.context.BaseContext;
import com.example.common.context.UserContext;
import com.example.common.enumeration.UserType;
import com.example.common.properties.JwtProperties;
import com.example.common.utils.JwtUtil;
import com.example.service.PermissionService;
import io.jsonwebtoken.Claims;
import jakarta.servlet.http.HttpServletRequest;
import jakarta.servlet.http.HttpServletResponse;
import lombok.extern.slf4j.Slf4j;
import org.springframework.beans.factory.annotation.Autowired;
import org.springframework.stereotype.Component;
import org.springframework.web.method.HandlerMethod;
import org.springframework.web.servlet.HandlerInterceptor;

/**
 * 统一JWT认证拦截器
 * 替换原有的三个独立拦截器
 */
@Component
@Slf4j
public class UnifiedJwtInterceptor implements HandlerInterceptor {
    
    @Autowired
    private JwtProperties jwtProperties;
    
    @Autowired
    private PermissionService permissionService;
    
    @Override
    public boolean preHandle(HttpServletRequest request, HttpServletResponse response, Object handler) throws Exception {
        // 判断是否为Controller方法
        if (!(handler instanceof HandlerMethod)) {
            return true;
        }
        
        // 根据路径判断用户类型
        String path = request.getRequestURI();
        UserType userType = UserType.fromPath(path);
        
        if (userType == null) {
            log.warn("无法识别的请求路径: {}", path);
            response.setStatus(401);
            return false;
        }
        
        // 获取Token
        String token = extractToken(request, userType);
        if (token == null) {
            log.warn("请求头中缺少Token, path: {}", path);
            response.setStatus(401);
            return false;
        }
        
        try {
            // 解析JWT
            String secretKey = getSecretKey(userType);
            Claims claims = JwtUtil.parseJWT(secretKey, token);
            
            // 构建用户上下文
            UserContext userContext = buildUserContext(claims, userType);
            
            // 如果是医生，查询角色和权限
            if (userType == UserType.DOCTOR) {
                enrichDoctorContext(userContext);
            } else if (userType == UserType.ADMIN) {
                enrichAdminContext(userContext);
            }
            
            // 存入上下文
            BaseContext.setCurrentUser(userContext);
            
            log.info("JWT校验成功, userType: {}, userId: {}, path: {}", 
                    userType, userContext.getUserId(), path);
            
            return true;
            
        } catch (Exception e) {
            log.error("JWT校验失败: {}", e.getMessage());
            response.setStatus(401);
            return false;
        }
    }
    
    /**
     * 从请求头中提取Token
     */
    private String extractToken(HttpServletRequest request, UserType userType) {
        String headerName;
        switch (userType) {
            case ADMIN:
                headerName = jwtProperties.getAdminTokenName();
                break;
            case DOCTOR:
                headerName = jwtProperties.getDoctorTokenName();
                break;
            case PATIENT:
            default:
                headerName = jwtProperties.getUserTokenName();
        }
        return request.getHeader(headerName);
    }
    
    /**
     * 获取密钥
```

```java
    @Override
    public List<String> getDoctorRoles(Long doctorId) {
        // 查询医生的所有角色
        LambdaQueryWrapper<DoctorRole> wrapper = new LambdaQueryWrapper<>();
        wrapper.eq(DoctorRole::getDoctorId, doctorId);
        List<DoctorRole> doctorRoles = doctorRoleMapper.selectList(wrapper);
        
        if (doctorRoles.isEmpty()) {
            return List.of("DOCTOR"); // 默认角色
        }
        
        // 查询角色编码
        List<Long> roleIds = doctorRoles.stream()
                .map(DoctorRole::getRoleId)
                .collect(Collectors.toList());
        
        return roleMapper.selectBatchIds(roleIds).stream()
                .map(SysRole::getRoleCode)
                .collect(Collectors.toList());
    }
    
    @Override
    public List<String> getDoctorPermissions(Long doctorId) {
        // 先获取角色列表
        List<String> roles = getDoctorRoles(doctorId);
        
        // 根据角色查询权限
        LambdaQueryWrapper<SysRole> roleWrapper = new LambdaQueryWrapper<>();
        roleWrapper.in(SysRole::getRoleCode, roles);
        List<SysRole> roleList = roleMapper.selectList(roleWrapper);
        
        List<Long> roleIds = roleList.stream()
                .map(SysRole::getId)
                .collect(Collectors.toList());
        
        if (roleIds.isEmpty()) {
            return List.of();
        }
        
        // 查询角色权限关联
        LambdaQueryWrapper<SysRolePermission> rpWrapper = new LambdaQueryWrapper<>();
        rpWrapper.in(SysRolePermission::getRoleId, roleIds);
        List<SysRolePermission> rolePermissions = rolePermissionMapper.selectList(rpWrapper);
        
        List<Long> permIds = rolePermissions.stream()
                .map(SysRolePermission::getPermissionId)
                .collect(Collectors.toList());
        
        // 查询权限编码
        return permissionMapper.selectBatchIds(permIds).stream()
                .map(SysPermission::getPermCode)
                .collect(Collectors.toList());
    }
    
    @Override
    public List<Long> getManagedDeptIds(Long doctorId) {
        // 查询医生是否为科室主任
        LambdaQueryWrapper<DoctorRole> wrapper = new LambdaQueryWrapper<>();
        wrapper.eq(DoctorRole::getDoctorId, doctorId);
        wrapper.eq(DoctorRole::getRoleId, getRoleIdByCode("DEPT_HEAD"));
        List<DoctorRole> roles = doctorRoleMapper.selectList(wrapper);
        
        return roles.stream()
                .map(DoctorRole::getDeptId)
                .collect(Collectors.toList());
    }
    
    @Override
    public List<String> getAdminRoles(Long adminId) {
        LambdaQueryWrapper<AdminRole> wrapper = new LambdaQueryWrapper<>();
        wrapper.eq(AdminRole::getAdminId, adminId);
        List<AdminRole> adminRoles = adminRoleMapper.selectList(wrapper);
        
        List<Long> roleIds = adminRoles.stream()
                .map(AdminRole::getRoleId)
                .collect(Collectors.toList());
        
        if (roleIds.isEmpty()) {
            return List.of("ADMIN");
        }
        
        return roleMapper.selectBatchIds(roleIds).stream()
                .map(SysRole::getRoleCode)
                .collect(Collectors.toList());
    }
    
    @Override
    public List<String> getAdminPermissions(Long adminId) {
        List<String> roles = getAdminRoles(adminId);
        
        // 超级管理员拥有所有权限
        if (roles.contains("SUPER_ADMIN")) {
            return permissionMapper.selectList(null).stream()
                    .map(SysPermission::getPermCode)
                    .collect(Collectors.toList());
        }
        
        // 普通管理员根据角色查权限
        return getDoctorPermissions(adminId); // 复用逻辑
    }
    
    @Override
    public boolean hasPermission(Long doctorId, String permission) {
        List<String> permissions = getDoctorPermissions(doctorId);
        return permissions.contains(permission);
    }
    
    @Override
    @Transactional
    public void assignRole(Long doctorId, Long roleId, Long deptId) {
        // 检查是否已存在
        LambdaQueryWrapper<DoctorRole> wrapper = new LambdaQueryWrapper<>();
        wrapper.eq(DoctorRole::getDoctorId, doctorId);
        wrapper.eq(DoctorRole::getRoleId, roleId);
        wrapper.eq(DoctorRole::getDeptId, deptId);
        
        if (doctorRoleMapper.selectCount(wrapper) > 0) {
            throw new RuntimeException("该角色已分配");
        }
        
        DoctorRole doctorRole = new DoctorRole();
        doctorRole.setDoctorId(doctorId);
        doctorRole.setRoleId(roleId);
        doctorRole.setDeptId(deptId);
        doctorRoleMapper.insert(doctorRole);
    }
    
    @Override
    @Transactional
    public void removeRole(Long doctorId, Long roleId, Long deptId) {
        LambdaQueryWrapper<DoctorRole> wrapper = new LambdaQueryWrapper<>();
        wrapper.eq(DoctorRole::getDoctorId, doctorId);
        wrapper.eq(DoctorRole::getRoleId, roleId);
        wrapper.eq(DoctorRole::getDeptId, deptId);
        doctorRoleMapper.delete(wrapper);
    }
    
    private Long getRoleIdByCode(String roleCode) {
        LambdaQueryWrapper<SysRole> wrapper = new LambdaQueryWrapper<>();
        wrapper.eq(SysRole::getRoleCode, roleCode);
        SysRole role = roleMapper.selectOne(wrapper);
        return role != null ? role.getId() : null;
    }
```

```java
package com.example.aspect;

import com.example.common.annotation.DataScope;
import com.example.common.annotation.RequirePermission;
import com.example.common.annotation.RequireRole;
import com.example.common.context.BaseContext;
import com.example.common.context.UserContext;
import lombok.extern.slf4j.Slf4j;
import org.aspectj.lang.ProceedingJoinPoint;
import org.aspectj.lang.annotation.Around;
import org.aspectj.lang.annotation.Aspect;
import org.springframework.stereotype.Component;

import java.util.Arrays;

/**
 * 权限检查切面
 */
@Aspect
@Component
@Slf4j
public class PermissionAspect {
    
    /**
     * 检查角色
     */
    @Around("@annotation(requireRole)")
    public Object checkRole(ProceedingJoinPoint point, RequireRole requireRole) throws Throwable {
        UserContext user = BaseContext.getCurrentUser();
        if (user == null) {
            throw new RuntimeException("用户未登录");
        }
        
        String[] roles = requireRole.value();
        boolean hasRole;
        
        if (requireRole.logical() == RequireRole.Logical.AND) {
            // 必须拥有所有角色
            hasRole = Arrays.stream(roles).allMatch(user::hasRole);
        } else {
            // 拥有任一角色即可
            hasRole = Arrays.stream(roles).anyMatch(user::hasRole);
        }
        
        if (!hasRole) {
            log.warn("用户 {} 缺少必要角色: {}", user.getUserId(), Arrays.toString(roles));
            throw new RuntimeException("无权访问");
        }
        
        return point.proceed();
    }
    
    /**
     * 检查权限
     */
    @Around("@annotation(requirePermission)")
    public Object checkPermission(ProceedingJoinPoint point, RequirePermission requirePermission) throws Throwable {
        UserContext user = BaseContext.getCurrentUser();
        if (user == null) {
            throw new RuntimeException("用户未登录");
        }
        
        // 超级管理员直接放行
        if (user.hasRole("SUPER_ADMIN")) {
            return point.proceed();
        }
        
        String[] permissions = requirePermission.value();
        boolean hasPermission;
        
        if (requirePermission.logical() == RequirePermission.Logical.AND) {
            hasPermission = Arrays.stream(permissions).allMatch(user::hasPermission);
        } else {
            hasPermission = Arrays.stream(permissions).anyMatch(user::hasPermission);
        }
        
        if (!hasPermission) {
            log.warn("用户 {} 缺少必要权限: {}", user.getUserId(), Arrays.toString(permissions));
            throw new RuntimeException("无权访问");
        }
        
        return point.proceed();
    }
}
```

#### 5.2.2 MyBatis数据权限拦截器

```java
package com.example.interceptor;

import com.baomidou.mybatisplus.core.toolkit.PluginUtils;
import com.example.common.annotation.DataScope;
import com.example.common.context.BaseContext;
import com.example.common.context.DataScopeContext;
import com.example.common.context.UserContext;
import lombok.extern.slf4j.Slf4j;
import net.sf.jsqlparser.expression.Expression;
import net.sf.jsqlparser.expression.LongValue;
import net.sf.jsqlparser.expression.operators.conditional.AndExpression;
import net.sf.jsqlparser.expression.operators.relational.EqualsTo;
import net.sf.jsqlparser.expression.operators.relational.ExpressionList;
import net.sf.jsqlparser.expression.operators.relational.InExpression;
import net.sf.jsqlparser.parser.CCJSqlParserUtil;
import net.sf.jsqlparser.schema.Column;
import net.sf.jsqlparser.statement.select.*;
import org.apache.ibatis.executor.statement.StatementHandler;
import org.apache.ibatis.mapping.BoundSql;
import org.apache.ibatis.mapping.MappedStatement;
import org.apache.ibatis.mapping.SqlCommandType;
import org.apache.ibatis.plugin.*;
import org.springframework.core.annotation.AnnotationUtils;
import org.springframework.stereotype.Component;
import org.springframework.web.method.HandlerMethod;

import java.lang.reflect.Method;
import java.sql.Connection;
import java.util.List;
import java.util.Properties;
import java.util.stream.Collectors;

/**
 * 数据权限拦截器
 * 自动在SQL中添加数据权限条件
 */
@Component
@Slf4j
@Intercepts({
    @Signature(type = StatementHandler.class, method = "prepare", 
               args = {Connection.class, Integer.class})
})
public class DataScopeInterceptor implements Interceptor {
    
    @Override
    public Object intercept(Invocation invocation) throws Throwable {
        StatementHandler statementHandler = (StatementHandler) PluginUtils.realTarget(invocation.getTarget());
        BoundSql boundSql = statementHandler.getBoundSql();
        
        // 获取MappedStatement
        MappedStatement mappedStatement = (MappedStatement) 
            invocation.getTarget().getClass().getDeclaredField("mappedStatement").get(statementHandler);
        
        // 只处理SELECT语句
        if (mappedStatement.getSqlCommandType() != SqlCommandType.SELECT) {
            return invocation.proceed();
        }
        
        // 获取方法上的@DataScope注解
        DataScope dataScope = getDataScopeAnnotation(mappedStatement);
        if (dataScope == null) {
            // 检查ThreadLocal中是否有数据权限设置
            dataScope = DataScopeContext.getDataScope();
        }
        
        if (dataScope == null) {
            return invocation.proceed();
        }
        
        // 获取当前用户
        UserContext user = BaseContext.getCurrentUser();
        if (user == null) {
            return invocation.proceed();
        }
        
        // 修改SQL
        String originalSql = boundSql.getSql();
        String modifiedSql = addDataScopeCondition(originalSql, dataScope, user);
        
        // 替换SQL
        PluginUtils.mpBoundSql(boundSql).sql(modifiedSql);
        
        return invocation.proceed();
    }
    
    /**
     * 获取方法上的@DataScope注解
     */
    private DataScope getDataScopeAnnotation(MappedStatement mappedStatement) {
        try {
            String id = mappedStatement.getId();
            String className = id.substring(0, id.lastIndexOf("."));
            String methodName = id.substring(id.lastIndexOf(".") + 1);
            
            Class<?> clazz = Class.forName(className);
            Method method = null;
            
            for (Method m : clazz.getMethods()) {
                if (m.getName().equals(methodName)) {
                    method = m;
                    break;
                }
            }
            
            if (method != null) {
                return AnnotationUtils.findAnnotation(method, DataScope.class);
            }
        } catch (Exception e) {
            log.warn("获取@DataScope注解失败: {}", e.getMessage());
        }
        return null;
    }
    
    /**
     * 添加数据权限条件
     */
    private String addDataScopeCondition(String sql, DataScope dataScope, UserContext user) {
        try {
            // 解析SQL
            net.sf.jsqlparser.statement.Statement statement = CCJSqlParserUtil.parse(sql);
            
            if (!(statement instanceof PlainSelect)) {
                return sql;
            }
            
            PlainSelect select = (PlainSelect) statement;
            Expression where = select.getWhere();
            
            // 构建数据权限条件
            Expression dataScopeCondition = buildDataScopeCondition(dataScope.type(), user);
            
            if (dataScopeCondition == null) {
                return sql;
            }
            
            // 合并条件
            if (where == null) {
                select.setWhere(dataScopeCondition);
            } else {
                select.setWhere(new AndExpression(where, dataScopeCondition));
            }
            
            return select.toString();
            
        } catch (Exception e) {
            log.error("添加数据权限条件失败: {}", e.getMessage());
            return sql;
        }
    }
    
    /**
     * 构建数据权限条件表达式
     */
    private Expression buildDataScopeCondition(DataScope.DataScopeType type, UserContext user) {
        switch (type) {
            case ALL:
                return null; // 不需要限制
                
            case OWN:
                // creator_id = userId
                return new EqualsTo(
                    new Column("creator_id"),
                    new LongValue(user.getUserId())
                );
                
            case DEPT:
                // dept_id = user.deptId
                return new EqualsTo(
                    new Column("dept_id"),
                    new LongValue(user.getDeptId())
                );
                
            case DEPT_AND_SUB:
                // dept_id IN (managedDeptIds)
                if (user.getManagedDeptIds() == null || user.getManagedDeptIds().isEmpty()) {
                    // 没有管理的科室，只能看本部门
                    return new EqualsTo(
                        new Column("dept_id"),
                        new LongValue(user.getDeptId())
                    );
                }
                
                List<Expression> deptIds = user.getManagedDeptIds().stream()
                    .map(LongValue::new)
                    .collect(Collectors.toList());
                
                return new InExpression()
                    .withLeftExpression(new Column("dept_id"))
                    .withRightExpression(new ExpressionList(deptIds));
                
            default:
                return null;
        }
    }
    
    @Override
    public Object plugin(Object target) {
        return Plugin.wrap(target, this);
    }
    
    @Override
    public void setProperties(Properties properties) {
    }
}
```
