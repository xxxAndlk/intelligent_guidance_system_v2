### 2.1 完整docker-compose.yml

```yaml
# deploy/docker-compose.yml
# 智慧医疗系统 - 完整基础设施配置

version: '3.8'

services:
  # ============================================
  # MySQL 8.0.X
  # ============================================
  mysql:
    image: mysql:8.0.36
    container_name: medical-mysql
    hostname: mysql
    environment:
      # 必须配置
      MYSQL_ROOT_PASSWORD: ${MYSQL_ROOT_PASSWORD:-root123}
      MYSQL_DATABASE: ${MYSQL_DATABASE:-medical_system}
      MYSQL_USER: ${MYSQL_USER:-medical}
      MYSQL_PASSWORD: ${MYSQL_PASSWORD:-medical123}
      
      # 时区配置
      TZ: Asia/Shanghai
      
      # MySQL 8.0特有配置
      MYSQL_AUTHENTICATION_PLUGIN: caching_sha2_password
    ports:
      - "${MYSQL_PORT:-3306}:3306"
    volumes:
      # 数据持久化
      - mysql_data:/var/lib/mysql
      # 初始化脚本
      - ./init/init.sql:/docker-entrypoint-initdb.d/01_init.sql:ro
      - ./init/seed.sql:/docker-entrypoint-initdb.d/02_seed.sql:ro
      # 配置文件
      - ./config/mysql/my.cnf:/etc/mysql/conf.d/my.cnf:ro
    networks:
      - medical-network
    command: >
      mysqld
      --default-authentication-plugin=caching_sha2_password
      --character-set-server=utf8mb4
      --collation-server=utf8mb4_unicode_ci
      --innodb_buffer_pool_size=512M
      --innodb_log_file_size=128M
      --max_connections=200
    healthcheck:
      test: ["CMD", "mysqladmin", "ping", "-h", "localhost", "-u", "root", "-p${MYSQL_ROOT_PASSWORD:-root123}"]
      interval: 10s
      timeout: 5s
      retries: 5
      start_period: 30s
    restart: unless-stopped
    logging:
      driver: "json-file"
      options:
        max-size: "100m"
        max-file: "3"

  # ============================================
  # Redis 7.4.X
  # ============================================
  redis:
    image: redis:7.4.0-alpine
    container_name: medical-redis
    hostname: redis
    environment:
      TZ: Asia/Shanghai
    ports:
      - "${REDIS_PORT:-6379}:6379"
    volumes:
      # 数据持久化
      - redis_data:/data
      # 配置文件
      - ./config/redis/redis.conf:/usr/local/etc/redis/redis.conf:ro
      # 初始化脚本
      - ./init/redis:/docker-entrypoint-initdb.d:ro
    networks:
      - medical-network
    command: redis-server /usr/local/etc/redis/redis.conf
    healthcheck:
      test: ["CMD", "redis-cli", "ping"]
      interval: 10s
      timeout: 3s
      retries: 5
    restart: unless-stopped
    logging:
      driver: "json-file"
      options:
        max-size: "50m"
        max-file: "3"

  # ============================================
  # RabbitMQ
  # ============================================
  rabbitmq:
    image: rabbitmq:3.13.0-management-alpine
    container_name: medical-rabbitmq
    hostname: rabbitmq
    environment:
      RABBITMQ_DEFAULT_USER: ${RABBITMQ_USER:-admin}
      RABBITMQ_DEFAULT_PASS: ${RABBITMQ_PASS:-admin123}
      RABBITMQ_DEFAULT_VHOST: ${RABBITMQ_VHOST:-medical}
      TZ: Asia/Shanghai
    ports:
      - "${RABBITMQ_AMQP_PORT:-5672}:5672"
      - "${RABBITMQ_MGMT_PORT:-15672}:15672"
    volumes:
      - rabbitmq_data:/var/lib/rabbitmq
      - ./config/rabbitmq/rabbitmq.conf:/etc/rabbitmq/rabbitmq.conf:ro
      - ./config/rabbitmq/definitions.json:/etc/rabbitmq/definitions.json:ro
    networks:
      - medical-network
    healthcheck:
      test: ["CMD", "rabbitmq-diagnostics", "ping"]
      interval: 10s
      timeout: 5s
      retries: 5
    restart: unless-stopped

  # ============================================
  # Qdrant 向量数据库
  # ============================================
  qdrant:
    image: qdrant/qdrant:v1.10.1
    container_name: medical-qdrant
    hostname: qdrant
    environment:
      QDRANT__SERVICE__HTTP_PORT: 6333
      QDRANT__SERVICE__GRPC_PORT: 6334
      QDRANT__STORAGE__STORAGE_PATH: /qdrant/storage
      QDRANT__STORAGE__SNAPSHOTS_PATH: /qdrant/snapshots
      QDRANT__LOG_LEVEL: INFO
      TZ: Asia/Shanghai
    ports:
      - "${QDRANT_HTTP_PORT:-6333}:6333"
      - "${QDRANT_GRPC_PORT:-6334}:6334"
    volumes:
      - qdrant_data:/qdrant/storage
      - qdrant_snapshots:/qdrant/snapshots
      - ./config/qdrant/config.yaml:/qdrant/config/production.yaml:ro
    networks:
      - medical-network
    healthcheck:
      test: ["CMD", "curl", "-f", "http://localhost:6333/healthz"]
      interval: 10s
      timeout: 5s
      retries: 5
    restart: unless-stopped

  # ============================================
  # Consul 服务注册
  # ============================================
  consul:
    image: consul:1.18.0
    container_name: medical-consul
    hostname: consul
    environment:
      CONSUL_BIND_INTERFACE: eth0
      TZ: Asia/Shanghai
    ports:
      - "${CONSUL_HTTP_PORT:-8500}:8500"
      - "${CONSUL_GRPC_PORT:-8600}:8600/tcp"
      - "${CONSUL_GRPC_PORT:-8600}:8600/udp"
      - "8300:8300"
      - "8301:8301/tcp"
      - "8301:8301/udp"
      - "8302:8302/tcp"
      - "8302:8302/udp"
    volumes:
      - consul_data:/consul/data
      - ./config/consul/consul-config.json:/consul/config/consul-config.json:ro
    networks:
      - medical-network
    command: >
      consul agent -dev -ui 
      -client=0.0.0.0 
      -bind='{{ GetInterfaceIP "eth0" }}'
      -config-file=/consul/config/consul-config.json
    healthcheck:
      test: ["CMD", "consul", "info"]
      interval: 10s
      timeout: 5s
      retries: 5
    restart: unless-stopped

  # ============================================
  # AI Service
  # ============================================
  ai-service:
    build:
      context: ../service/ai
      dockerfile: Dockerfile
    container_name: medical-ai-service
    environment:
      # 服务配置
      - SERVER_HTTP_ADDR=0.0.0.0:8001
      - SERVER_GRPC_ADDR=0.0.0.0:9001
      
      # MySQL配置
      - DATA_DATABASE_DRIVER=mysql
      - DATA_DATABASE_HOST=mysql
      - DATA_DATABASE_PORT=3306
      - DATA_DATABASE_USER=${MYSQL_USER:-medical}
      - DATA_DATABASE_PASSWORD=${MYSQL_PASSWORD:-medical123}
      - DATA_DATABASE_NAME=${MYSQL_DATABASE:-medical_system}
      - DATA_DATABASE_SOURCE=${MYSQL_USER:-medical}:${MYSQL_PASSWORD:-medical123}@tcp(mysql:3306)/${MYSQL_DATABASE:-medical_system}?parseTime=True&loc=Local&charset=utf8mb4
      
      # Redis配置
      - DATA_REDIS_ADDR=redis:6379
      - DATA_REDIS_PASSWORD=${REDIS_PASSWORD:-}
      - DATA_REDIS_DB=${REDIS_DB:-0}
      - DATA_REDIS_POOL_SIZE=${REDIS_POOL_SIZE:-10}
      - DATA_REDIS_MIN_IDLE=${REDIS_MIN_IDLE:-5}
      - DATA_REDIS_MAX_RETRIES=${REDIS_MAX_RETRIES:-3}
      
      # Qdrant配置
      - VECTOR_DB_HOST=qdrant
      - VECTOR_DB_GRPC_PORT=6334
      - VECTOR_DB_HTTP_PORT=6333
      - VECTOR_DB_TIMEOUT=10s
      
      # RabbitMQ配置
      - MQ_HOST=rabbitmq
      - MQ_PORT=5672
      - MQ_USER=${RABBITMQ_USER:-admin}
      - MQ_PASSWORD=${RABBITMQ_PASS:-admin123}
      - MQ_VHOST=${RABBITMQ_VHOST:-medical}
      
      # Consul配置
      - REGISTRY_CONSUL_ADDR=consul:8500
      - SERVICE_NAME=ai-service
      - SERVICE_VERSION=v1.0.0
      
      # LLM配置
      - LLM_API_KEY=${SILICONFLOW_API_KEY}

      - LLM_MODEL=Pro/deepseek-ai/DeepSeek-V3
```

## 四、数据库初始化脚本

### 4.1 MySQL初始化脚本 init.sql

```sql
-- deploy/init/init.sql
-- MySQL 8.0.X 初始化脚本

-- 设置字符集
SET NAMES utf8mb4;
SET FOREIGN_KEY_CHECKS = 0;

-- ============================================
-- 创建数据库
-- ============================================
CREATE DATABASE IF NOT EXISTS medical_system 
DEFAULT CHARACTER SET utf8mb4 
DEFAULT COLLATE utf8mb4_unicode_ci;

USE medical_system;

-- ============================================
-- 用户表
-- ============================================
CREATE TABLE IF NOT EXISTS `user` (
    `id` BIGINT UNSIGNED NOT NULL AUTO_INCREMENT COMMENT '用户ID',
    `openid` VARCHAR(255) DEFAULT NULL COMMENT '微信OpenID',
    `username` VARCHAR(50) NOT NULL COMMENT '用户名',
    `password` VARCHAR(255) NOT NULL COMMENT '加密密码',
    `name` VARCHAR(50) DEFAULT NULL COMMENT '姓名',
    `age` INT DEFAULT NULL COMMENT '年龄',
    `phone` VARCHAR(20) DEFAULT NULL COMMENT '手机号',
    `sex` TINYINT DEFAULT NULL COMMENT '性别：1男 2女',
    `email` VARCHAR(100) DEFAULT NULL COMMENT '邮箱',
    `id_card` VARCHAR(18) DEFAULT NULL COMMENT '身份证号',
    `avatar` VARCHAR(500) DEFAULT NULL COMMENT '头像URL',
    `home_address` VARCHAR(255) DEFAULT NULL COMMENT '家庭地址',
    `status` TINYINT DEFAULT 1 COMMENT '状态：0禁用 1启用',
    `created_at` DATETIME DEFAULT CURRENT_TIMESTAMP COMMENT '创建时间',
    `updated_at` DATETIME DEFAULT CURRENT_TIMESTAMP ON UPDATE CURRENT_TIMESTAMP COMMENT '更新时间',
    `deleted_at` DATETIME DEFAULT NULL COMMENT '删除时间',
    PRIMARY KEY (`id`),
    UNIQUE KEY `uk_username` (`username`),
    UNIQUE KEY `uk_phone` (`phone`),
    KEY `idx_status` (`status`),
    KEY `idx_deleted` (`deleted_at`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4 COLLATE=utf8mb4_unicode_ci COMMENT='用户表';

-- ============================================
-- 医生表
```

```go
// deploy/init/vector_init.go
// Qdrant向量数据库初始化

package main

import (
    "context"
    "fmt"
    "log"
    
    "github.com/qdrant/go-client/qdrant"
)

func main() {
    ctx := context.Background()
    
    // 连接Qdrant
    client, err := qdrant.NewClient(&qdrant.Config{
        Host: "localhost",
        Port: 6334,
    })
    if err != nil {
        log.Fatal(err)
    }
    
    // 创建diseases集合
    if err := createCollection(ctx, client, "diseases", 1024); err != nil {
        log.Printf("创建diseases集合失败: %v", err)
    }
    
    // 创建doctors集合
    if err := createCollection(ctx, client, "doctors", 1024); err != nil {
        log.Printf("创建doctors集合失败: %v", err)
    }
    
    // 创建drugs集合
    if err := createCollection(ctx, client, "drugs", 1024); err != nil {
        log.Printf("创建drugs集合失败: %v", err)
    }
    
    // 创建medical_records集合
    if err := createCollection(ctx, client, "medical_records", 1024); err != nil {
        log.Printf("创建medical_records集合失败: %v", err)
    }
    
    fmt.Println("向量数据库初始化完成")
}

func createCollection(ctx context.Context, client *qdrant.Client, name string, vectorSize uint64) error {
    // 检查集合是否已存在
    exists, err := client.CollectionExists(ctx, name)
    if err != nil {
        return err
    }
    
    if exists {
        fmt.Printf("集合 %s 已存在，跳过创建\n", name)
        return nil
    }
    
    // 创建集合
    _, err = client.CreateCollection(ctx, &qdrant.CreateCollection{
        CollectionName: name,
        VectorsConfig: &qdrant.VectorsConfig{
            Config: &qdrant.VectorsConfig_Params{
                Params: &qdrant.VectorParams{
                    Size:     vectorSize,
                    Distance: qdrant.Distance_Cosine,
                },
            },
        },
    })
    
    if err != nil {
        return err
    }
    
    fmt.Printf("集合 %s 创建成功\n", name)
    return nil
}
```
